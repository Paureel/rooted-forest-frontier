import DisjointExact
namespace OrdinaryDispatch.Hard2Combine
open RMAF ForestState OutputState RestrictionBridge Universal233 Universal233.CherryBranching Universal233.BothBad
abbrev Ordinary := DisjointParents.Ordinary

def unitCalls (S : Tree) (T : Tree) (a b q : Nat) : List Call :=
 [singletonChild S a,singletonChild S b].filterMap (fun c=>
  (remaining q c).map (fun r=>⟨some T,c.sourceForest,r⟩))

def combine (actual : Ordinary) (T S : Tree) (a b q : Nat) (paid : Execution) : Execution :=
 ⟨choose (runChildren actual T S q [singletonChild S a,singletonChild S b]) paid.answer,
  unitCalls S T a b q++paid.calls⟩

theorem combine_sound (actual : Ordinary) (T S : Tree) (a b q : Nat) (paid : Execution)
 (sound : ∀F,paid.answer=some F→agreement T S F=true ∧ F.length-1≤q)
 (F : OrdinaryDispatch.Forest) (hf : (combine actual T S a b q paid).answer=some F) :
 agreement T S F=true ∧ F.length-1≤q := by
 rcases choose_origin _ _ F hf with h|h
 · exact runChildren_sound actual T S q _ F h
 · exact sound F h

theorem combine_complete (actual : Ordinary) (T S : Tree) (a b q : Nat) (paid : Execution)
 {p : RMAF.Vertex} (nt : (leaves T).Nodup) (ns : (leaves S).Nodup)
 (same : ∀x,x∈leaves T ↔ x∈leaves S) (ch : CoreGraph.CherryAt T p a b)
 (ih : ∀r,r<q→ForestMinCompletion.LocalOrdinaryIH r actual)
 (parent : ∀F,agreement T S F=true→Refines F (pendantPartition S a b)→F.length-1≤q→
  ∃G,paid.answer=some G ∧ G.length≤F.length)
 (F : OrdinaryDispatch.Forest) (hf : agreement T S F=true) (budget : F.length-1≤q) :
 ∃G,(combine actual T S a b q paid).answer=some G ∧ G.length≤F.length := by
 obtain ⟨c,hc,hfeas⟩ := ordinary_branches_cover ns hf ch
 have cases : c=singletonChild S a ∨ c=singletonChild S b ∨ c=pendantChild S a b := by
  simpa [branches] using hc
 rcases cases with ha|hb|hp
 · subst c
   obtain ⟨part,emb,paidId,mat⟩ := three_children_literal nt ns same ch _ hc
   obtain ⟨H,hH,hmin⟩ := DisjointParents.member_complete actual T S q [singletonChild S a,singletonChild S b]
    nt ns same _ (by simp) part emb paidId mat
    (fun r hr=>ih r (remaining_decreases (Nat.le_refl 1) hr)) F hfeas budget
   obtain ⟨G,hG,hGH⟩ := choose_left_min _ _ H hH
   exact ⟨G,hG,Nat.le_trans hGH hmin⟩
 · subst c
   obtain ⟨part,emb,paidId,mat⟩ := three_children_literal nt ns same ch _ hc
   obtain ⟨H,hH,hmin⟩ := DisjointParents.member_complete actual T S q [singletonChild S a,singletonChild S b]
    nt ns same _ (by simp) part emb paidId mat
    (fun r hr=>ih r (remaining_decreases (Nat.le_refl 1) hr)) F hfeas budget
   obtain ⟨G,hG,hGH⟩ := choose_left_min _ _ H hH
   exact ⟨G,hG,Nat.le_trans hGH hmin⟩
 · subst c
   obtain ⟨H,hH,hmin⟩ := parent F hf hfeas.2 budget
   obtain ⟨G,hG,hGH⟩ := choose_right_min _ _ H hH
   exact ⟨G,hG,Nat.le_trans hGH hmin⟩

theorem unitCalls_strict (T S : Tree) (a b q : Nat) (c : Call)
 (hc : c∈unitCalls S T a b q) : c.budget<q := by
 obtain ⟨child,mem,eq⟩ := List.mem_filterMap.1 hc
 obtain ⟨r,hr,he⟩ := Option.map_eq_some_iff.1 eq
 have cost : child.paid=1 := by
  have choices : child=singletonChild S a ∨ child=singletonChild S b := by simpa using mem
  rcases choices with rfl|rfl <;> rfl
 have drop := remaining_decreases (by omega : 1≤child.paid) hr
 have ceq : c=⟨some T,child.sourceForest,r⟩ := he.symm
 subst c
 exact drop

theorem unitCalls_mass (T S : Tree) (a b q : Nat) (c : Rat) (hc : 2≤c) :
 traceMass c q (unitCalls S T a b q)≤2*c^(-1 : Int) := by
 by_cases affordable : 1≤q
 · have sub : q-(q-1)=1 := by omega
   simp [unitCalls,remaining,singletonChild,makeChild,affordable,traceMass,
    PaidQueries.actualQueryMass,sub,Rat.add_zero,Rat.two_mul]
 · have no : ¬(singletonChild S a).paid≤q := by change ¬1≤q;exact affordable
   have no' : ¬(singletonChild S b).paid≤q := by change ¬1≤q;exact affordable
   simp only [unitCalls,List.filterMap_cons,remaining,no,no',ite_false,Option.map_none,
    List.filterMap_nil,traceMass,PaidQueries.actualQueryMass,List.map_nil,List.sum_nil]
   exact Rat.mul_nonneg (by decide) (Rat.le_of_lt (Rat.zpow_pos (by grind : 0<c)))

theorem combine_mass (actual : Ordinary) (T S : Tree) (a b q : Nat) (paid : Execution)
 (c : Rat) (hc : 2≤c) (v : Rat) (bound : traceMass c q paid.calls≤v) :
 traceMass c q (combine actual T S a b q paid).calls≤2*c^(-1 : Int)+v := by
 rw [combine,traceMass_append]
 exact SourceCoverage.qadd_le_add (unitCalls_mass T S a b q c hc) bound
end OrdinaryDispatch.Hard2Combine
