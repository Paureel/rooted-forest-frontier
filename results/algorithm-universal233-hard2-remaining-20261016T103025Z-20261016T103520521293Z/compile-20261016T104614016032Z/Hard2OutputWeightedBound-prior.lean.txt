theorem case_output_generated (S T : Tree) (A B C : List Nat) (q : Nat) (out : Result) (o : Output)
 (mem : o∈caseOutputs S T A B C q out) :
 ∃color:Nat→Nat,o∈(generate S color q).outputs := by
 have origin (kind : Kind) (A B : List Nat) (h:o∈outputs kind S T A B q) :
  ∃color:Nat→Nat,o∈(generate S color q).outputs:=by
  obtain ⟨ca,cb,cc,h⟩:=outputs_generated kind S T A B q o h
  exact ⟨_,h⟩
 cases out with
 | disjoint=>simp [caseOutputs] at mem
 | clique v=>exact origin (.clique v) A B mem
 | four order v w=>exact origin (.four v w) _ _ mem
 | diagonal order v w=>exact origin (.diagonal v w) _ _ mem
 | pair order k=>cases k with
  | vertex v=>exact origin (.vertex v) _ _ mem
  | edge e=>
   simp only [caseOutputs,edgeOutputs] at mem
   split at mem <;> exact origin (.side _) _ _ mem

theorem actual_output_generated (S T : Tree) (c : CherryBranching.Candidate) (q : Nat) (out : Result) (o : Output)
 (mem : o∈actualOutputs S T c q out) : ∃color:Nat→Nat,o∈(generate S color q).outputs := by
 unfold actualOutputs at mem
 split at mem
 · exact case_output_generated S T _ _ _ q out o mem
 · simp at mem

theorem actual_output_shape (S T : Tree) (c : CherryBranching.Candidate) (q : Nat) (out : Result) (N : Nat)
 (ns : (leaves S).Nodup) (sn : treeNodes S≤N) (o : Output) (mem : o∈actualOutputs S T c q out) :
 partitionCheck (leaves S) o.partition=true ∧ o.sourceCuts.length≤N ∧ o.partition.length≤N ∧
 (∀B,B∈o.partition→B.length≤N) := by
 obtain ⟨color,gen⟩:=actual_output_generated S T c q out o mem
 have valid:=(Assembly.generated_source_valid S color q o gen).1
 have shape:=RankDichotomy.generate_partition_cost S color q o gen
 have blocks:=SourceCoverage.partitionCheck_block_bound _ _ valid
 have flat:=(FinalCutRecovery.checked_flatten_perm S o.partition ns valid).length_eq
 have sl:=leaves_nodes S
 refine ⟨valid,by omega,by omega,?_⟩
 intro B hB
 have b:=block_cells o.partition B hB
 omega

/-- Weighted forest debit: the forest overhead at q-d is multiplied by the
actual raw output weight c^-d. Failed and duplicate occurrences remain in sum. -/
theorem outputForestDebit_weighted (actual : GenericForest.FrontierWork.Ordinary)
 (T S : Tree) (q N Q : Nat) (os : List Output)
 (tl : (leaves T).length≤N) (budget : q≤Q)
 (proper0 : ∀o,o∈os→proper (some T) (OutputState.sourceForest S o.partition)=true) :
 (outputForestDebit actual (fun _ _ _=>0) T S q os:Rat)≤
 (geometric Rule5ActualTrace.base*forestCoefficient N Q*Rule5ActualTrace.base^q)*rawQMass Rule5ActualTrace.base os := by
 let b:=Rule5ActualTrace.base
 let K:Rat:=geometric b*forestCoefficient N Q
 have kp:0≤K:=Rat.mul_nonneg (geometric_properties b (by decide +kernel)).2.2.1 Rat.natCast_nonneg
 have bp:0<b:=by decide +kernel
 have point (o : Output) (mem : o∈os) :
 ((if o.sourceCuts.length≤q then forestWork actual (fun _ _ _=>0) (some T)
  (OutputState.sourceForest S o.partition) (q-o.sourceCuts.length) else 0:Nat):Rat)≤
 (K*b^q)*b^(-(o.sourceCuts.length:Int)):=by
  by_cases h:o.sourceCuts.length≤q
  · simp only [h,ite_true]
    have bill:=forestLocalWork_bound actual (some T) N Q b (by decide +kernel) tl
     (OutputState.sourceForest S o.partition) (q-o.sourceCuts.length) (by omega) (proper0 o mem)
    rw [qpow_clear b (Rat.ne_of_gt bp) q _ h] at bill
    simpa only [K,b,Rat.mul_assoc] using bill
  · simp only [h,ite_false,Rat.natCast_ofNat]
    exact Rat.mul_nonneg (Rat.mul_nonneg kp (Rat.le_of_lt (Rat.pow_pos bp))) (Rat.le_of_lt (Rat.zpow_pos bp))
 have sum:=GenericForest.rat_sum_map_le os _ _ point
 rw [GenericForest.rat_sum_mul] at sum
 simpa only [outputForestDebit,nat_sum_cast,rawQMass,K,b] using sum

theorem kind_output_count (S T : Tree) (A B : List Nat) (q : Nat) (kind : Kind) :
 (outputs kind S T A B q).length≤4*outputCountCap S q := by
 have call (ca cb cc : Nat→Nat) : (wholeOutputs S A B ca cb cc q).length≤outputCountCap S q:=
  generate_output_count S _ q
 cases kind with
 | clique v=>have a:=call (fun _=>0) (colorLabel T v) (colorLabel T v)
             have b:=call (colorLabel T v) (fun _=>0) (colorLabel T v)
             have c:=call (colorLabel T v) (colorLabel T v) (fun _=>0)
             simp only [outputs,cliqueOutputs,List.length_append];omega
 | four v w=>have a:=call (fun _=>0) (colorLabel T v) (colorLabel T w)
             have b:=call (fun _=>0) (jointLabel T v w) (fun _=>0)
             have c:=call (colorLabel T v) (fun _=>0) (colorLabel T w)
             have d:=call (colorLabel T v) (colorLabel T w) (fun _=>0)
             simp only [outputs,fourChainOutputs,List.length_append];omega
 | diagonal v w=>have a:=call (fun _=>0) (jointLabel T v w) (fun _=>0)
                 have b:=call (colorLabel T v) (fun _=>0) (colorLabel T w)
                 simp only [outputs,diagonalOutputs,List.length_append];omega
 | vertex v=>have a:=call (colorLabel T v) (fun _=>0) (fun _=>0)
             have b:=call (fun _=>0) (colorLabel T v) (fun _=>0)
             simp only [outputs,PairAssembly.vertexPairOutputs,List.length_append];omega
 | side D=>have a:=call (fun _=>0) (PairAssembly.sideColor D) (fun _=>0)
           simp only [outputs,PairAssembly.sideOutputs];omega

theorem case_output_count (S T : Tree) (A B C : List Nat) (q : Nat) (out : Result) :
 (caseOutputs S T A B C q out).length≤4*outputCountCap S q := by
 cases out with
 | disjoint=>simp [caseOutputs]
 | clique v=>exact kind_output_count S T A B q (.clique v)
 | four o v w=>exact kind_output_count S T _ _ q (.four v w)
 | diagonal o v w=>exact kind_output_count S T _ _ q (.diagonal v w)
 | pair order k=>cases k with
  | vertex v=>exact kind_output_count S T _ _ q (.vertex v)
  | edge e=>simp only [caseOutputs,edgeOutputs];split <;> exact kind_output_count S T _ _ q (.side _)

theorem actual_output_count (S T : Tree) (c : CherryBranching.Candidate) (q : Nat) (out : Result) :
 (actualOutputs S T c q out).length≤12*2^(q+1) := by
 have power:=Nat.pow_le_pow_right (show 1≤2 by decide) (show min q ((leaves S).length-1)+1≤q+1 by omega)
 unfold actualOutputs
 split
 · have h:=case_output_count S T _ _ _ q out
   unfold outputCountCap at h
   omega
 · simp
