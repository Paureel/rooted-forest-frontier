def ResultSize (N : Nat) : Result→Prop
 | .disjoint=>True
 | .clique v=>v.length≤N
 | .four _ v w=>v.length≤N ∧ w.length≤N
 | .diagonal _ v w=>v.length≤N ∧ w.length≤N
 | .pair _ (.vertex v)=>v.length≤N
 | .pair _ (.edge e)=>e.2.length≤N

theorem selected_result_size {T : Tree} {A B C : List Nat} {out : Result}
 (ht : (leaves T).Nodup) (hl : ∀x,x∈leaves S→x∈leaves T)
 (hp : partitionCheck (leaves S) [A,B,C]=true) (sel : Hard2Overlap.select T A B C=some out) :
 ResultSize (treeNodes T) out := by
 have h:=Hard2Overlap.select_sound ht hl hp sel
 have addr (D : List Nat) (v : RMAF.Vertex) (mem : v∈minimalVertices D T) : v.length≤treeNodes T:=by
  have h:=minimal_address D T v mem;have l:=leaves_nodes T;omega
 cases out with
 | disjoint=>trivial
 | clique v=>exact addr A v h.2.1
 | four o v w=>exact ⟨addr _ v h.1.2.2.1,addr _ w h.1.2.2.2.2.1⟩
 | diagonal o v w=>exact ⟨addr _ v h.1.2.2.1,addr _ w h.1.2.2.2.2.1⟩
 | pair o k=>cases k with
  | vertex v=>exact addr _ v h.2.2.2.1
  | edge e=>exact (pairEdges_address T e h.2.2.2.1).2

theorem caseOutputsMeter_bound (S T : Tree) (A B C : List Nat) (N q : Nat) (three : 3≤N)
 (tn : treeNodes T≤N) (sn : treeNodes S≤N) (al : A.length≤N) (bl : B.length≤N) (cl : C.length≤N)
 (rs : ResultSize N out) :
 (caseOutputsMeter S T A B C q out).work≤outputKindCap S T N q+200*(N+1)^3 ∧
 (caseOutputsMeter S T A B C q out).value.length≤4*outputCountCap S q := by
 have firstCap (o : Order) : (first o A B C).length≤N:=by cases o <;> assumption
 have secondCap (o : Order) : (second o A B C).length≤N:=by cases o <;> assumption
 have kind (o : Order) (k : Kind) (size : KindSize N k):=
  kindOutputsMeter_bound k S T (first o A B C) (second o A B C) N q (firstCap o) (secondCap o) size
 cases out with
 | disjoint=>simp [caseOutputsMeter,outputKindCap];omega
 | clique v=>have h:=kindOutputsMeter_bound (.clique v) S T A B N q al bl rs
             simp only [caseOutputsMeter];constructor <;> omega
 | four o v w=>have h:=kind o (.four v w) rs;simp only [caseOutputsMeter];constructor <;> omega
 | diagonal o v w=>have h:=kind o (.diagonal v w) rs;simp only [caseOutputsMeter];constructor <;> omega
 | pair o k=>cases k with
  | vertex v=>have h:=kind o (.vertex v) rs;simp only [caseOutputsMeter];constructor <;> omega
  | edge e=>
   have dl : (pairDescendantsMeter T e.2).value.length≤N:=by
    rw [pairDescendantsMeter_value]
    exact Nat.le_trans (List.length_filter_le _ _) (Nat.le_trans (leaves_nodes T) tn)
   have a:=kind o (.side (pairDescendantsMeter T e.2).value) dl
   have b:=kindOutputsMeter_bound (.side (pairDescendantsMeter T e.2).value) S T
    (second o A B C) (first o A B C) N q (secondCap o) (firstCap o) dl
   have d:=pairDescendantsMeter_bound T e.2 N tn rs
   have check:=meterRestrictionAgrees_bound S T (first o A B C) N (by omega) (by omega) (firstCap o)
   have lift:=cap_lift N 128 2 3 _ check (by decide)
   have pos:1≤(N+1)^3:=Nat.one_le_pow _ _ (by omega)
   by_cases good:(meterRestrictionAgrees S T (first o A B C)).value=true
   · simp only [caseOutputsMeter,edgeOutputsMeter,good,ite_true];constructor <;> omega
   · simp only [caseOutputsMeter,edgeOutputsMeter,good,Bool.false_eq_true,ite_false];constructor <;> omega

def generationPolynomial (N : Nat) :=128*(N+1)^3+16*(1024*(N*(N+2)+N+2)^3+6)+12

theorem generationEnvelope_uniform (S : Tree) (N q : Nat) (sn : treeNodes S≤N) :
 generationEnvelope S q≤generationPolynomial N*(q+1)*2^(q+1) := by
 have init:initialCharge S≤128*(N+1)^3:=by
  unfold initialCharge;exact Nat.mul_le_mul_left 128 (Nat.pow_le_pow_left (by omega) 3)
 have rc:inputResource S+1≤N*(N+2)+N+2:=by
  have h:=Nat.mul_le_mul sn (show treeNodes S+2≤N+2 by omega)
  unfold inputResource;omega
 have lc:RecoveryLocalWork.localCap (inputResource S)≤1024*(N*(N+2)+N+2)^3:=
  Nat.mul_le_mul_left 1024 (Nat.pow_le_pow_left rc 3)
 have mq:min q ((leaves S).length-1)+1≤q+1:=by omega
 have power:=Nat.pow_le_pow_right (show 1≤2 by decide) mq
 have poly:=Nat.mul_le_mul (Nat.mul_le_mul (Nat.add_le_add_right lc 6) mq) power
 have pos:1≤(q+1)*2^(q+1):=Nat.mul_pos (by omega) (Nat.two_pow_pos _)
 have rest:=Nat.mul_le_mul_left (128*(N+1)^3+12) pos
 simp only [generationEnvelope,generationPolynomial,Nat.add_mul,Nat.mul_add,Nat.one_mul,Nat.mul_one,Nat.mul_assoc] at *
 omega

def taggedPolynomial (N : Nat) :=8*(N+1)+3*(2*(16*(N+3)*N+4*(N+1)+3)+3)+8

theorem uniformTaggedCap_mono (T : Tree) (N : Nat) (tn : treeNodes T≤N) :
 uniformTaggedCap T N≤taggedPolynomial N := by
 unfold uniformTaggedCap localAddressCap taggedPolynomial
 repeat first | exact tn | exact Nat.le_refl _ | apply Nat.add_le_add | apply Nat.mul_le_mul

def generationCallPolynomial (N : Nat) :=generationPolynomial N+N*taggedPolynomial N

theorem outputKindCap_uniform (S T : Tree) (N q : Nat) (tn : treeNodes T≤N) (sn : treeNodes S≤N) :
 outputKindCap S T N q≤(4*generationCallPolynomial N+52)*(q+1)*2^(q+1) := by
 let f:=(q+1)*2^(q+1)
 have pos:1≤f:=Nat.mul_pos (by omega) (Nat.two_pow_pos _)
 have gen:=generationEnvelope_uniform S N q sn
 have color:=Nat.mul_le_mul (Nat.le_trans (leaves_nodes S) sn) (uniformTaggedCap_mono T N tn)
 have colorLift:=Nat.mul_le_mul_left (N*taggedPolynomial N) pos
 have power:=Nat.pow_le_pow_right (show 1≤2 by decide) (show min q ((leaves S).length-1)+1≤q+1 by omega)
 have powLift:2^(q+1)≤f:=by simpa only [Nat.one_mul] using Nat.mul_le_mul_right (2^(q+1)) (show 1≤q+1 by omega)
 have count:outputCountCap S q≤3*f:=by unfold outputCountCap;omega
 change generationEnvelope S q≤generationPolynomial N*f at gen
 have call:outputCallCap S T N q≤generationCallPolynomial N*f:=by
  unfold outputCallCap generationCallPolynomial
  rw [Nat.add_mul]
  omega
 change outputKindCap S T N q≤(4*generationCallPolynomial N+52)*f
 unfold outputKindCap
 have four:=Nat.mul_le_mul_left 4 call
 have twelve:=Nat.mul_le_mul_left 12 count
 have constant:=Nat.mul_le_mul_left 16 pos
 simp only [Nat.add_mul,Nat.mul_assoc] at *
 omega

def hard2GenerationPolynomial (N : Nat) :=4*generationCallPolynomial N+52+262*(N+1)^3+8

theorem actualOutputsMeter_bound (T S : Tree) (c : CherryBranching.Candidate) (out : Result) (N q : Nat)
 (three : 3≤N) (tn : treeNodes T≤N) (sn : treeNodes S≤N)
 (proper0 : DepthOne.proper T S=true) (hard : CherryBranching.select T S=some (.hard2,c))
 (selected : Hard2Overlap.runHard2 T S c=some out) :
 (actualOutputsMeter S T c q out).work≤hard2GenerationPolynomial N*(q+1)*2^(q+1) := by
 obtain ⟨p,r,cuts,_,_,hp,_,_⟩:=Hard2Overlap.actual_source_inputs proper0 hard
 have ht:=((DepthOne.proper_iff T S).1 proper0).1
 have same:=((DepthOne.proper_iff T S).1 proper0).2.2
 have sel:Hard2Overlap.select T (leftParent S p) (rightParent S r) (rootParent S p r)=some out:=by
  simpa only [Hard2Overlap.runHard2,cuts] using selected
 have sizes:=selected_result_size ht (fun x hx=>same.mem_iff.2 hx) hp sel
 have rs:ResultSize N out:=by
  cases out with
  | pair o k=>cases k <;> dsimp only [ResultSize] at * <;> omega
  | _=>dsimp only [ResultSize] at * <;> omega
 have a:(leftParent S p).length≤N:=Nat.le_trans (subtree_labels_payload S p).1 (Nat.le_trans (leaves_nodes S) sn)
 have b:(rightParent S r).length≤N:=Nat.le_trans (subtree_labels_payload S r).1 (Nat.le_trans (leaves_nodes S) sn)
 have d:(rootParent S p r).length≤N:=Nat.le_trans (List.length_filter_le _ _) (Nat.le_trans (leaves_nodes S) sn)
 have body:=caseOutputsMeter_bound S T (leftParent S p) (rightParent S r) (rootParent S p r) N q three tn sn a b d rs
 have parents:=hard2ParentsMeter_bound S p r N sn
 have uniform:=outputKindCap_uniform S T N q tn sn
 have pos:1≤(q+1)*2^(q+1):=Nat.mul_pos (by omega) (Nat.two_pow_pos _)
 have lift:=Nat.mul_le_mul_left (262*(N+1)^3+8) pos
 simp only [actualOutputsMeter,cuts,hard2ParentsMeter_value]
 simp only [hard2GenerationPolynomial,Nat.add_mul,Nat.mul_assoc] at *
 omega
end OrdinaryDispatch.Hard2SourceWork
