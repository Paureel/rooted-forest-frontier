def forestWork (ordinary : Ordinary) (W : OrdinaryWork) (T : Option Tree) (ss : List Tree) (q : Nat) : Nat :=
 nonrecursiveBill ordinary W T ss q+
 if proper T ss && !agrees T ss (ss.map leaves) then
  match T,ss with
  | none,_ => 0
  | some _,[_] => 0
  | some t,_ => match overlap t ss with
    | none => 0
    | some h => ((branches t ss h q).map fun io=>
     let d := io.2.sourceCuts.length
     if _hd : 1≤d ∧ d≤q then forestWork ordinary W T (replace ss io.1 io.2.partition) (q-d)
     else 0).sum
 else 0
termination_by q
decreasing_by omega

theorem nat_sum_cast {α : Type} (L : List α) (f : α→Nat) :
 ((L.map f).sum:Rat)=(L.map fun a=>(f a:Rat)).sum := by
 induction L with
 | nil => rfl
 | cons a L ih => simp only [List.map_cons,List.sum_cons,Rat.natCast_add,ih]

/-- Exact integer accounting of the frozen rational source sheet. It includes
all actual recursive filterMap occurrences, even failed and duplicate calls. -/
theorem forestWork_cast (ordinary : Ordinary) (W : OrdinaryWork) (T : Option Tree) (ss : List Tree) (q : Nat) :
 (forestWork ordinary W T ss q:Rat)=GenericForest.sourceWork ordinary W T ss q := by
 induction q using Nat.strongRecOn generalizing ss with
 | ind q ih =>
  have child (t : Tree) (h : Overlap) :
   (((branches t ss h q).map fun io=>if _hd:1≤io.2.sourceCuts.length ∧ io.2.sourceCuts.length≤q then
    forestWork ordinary W T (replace ss io.1 io.2.partition) (q-io.2.sourceCuts.length) else 0).sum:Rat)=
   ((branches t ss h q).map fun io=>if _hd:1≤io.2.sourceCuts.length ∧ io.2.sourceCuts.length≤q then
    sourceWork ordinary W T (replace ss io.1 io.2.partition) (q-io.2.sourceCuts.length) else 0).sum := by
   rw [nat_sum_cast]
   congr 1
   apply List.map_congr_left
   intro io _
   split
   · rename_i hd
     simpa only [dite_eq_left hd] using ih (q-io.2.sourceCuts.length) (by omega) (replace ss io.1 io.2.partition)
   · rename_i hd
     rfl
  cases T with
  | none => simp [forestWork.eq_def,GenericForest.sourceWork.eq_def,Rat.natCast_add,Rat.add_zero]
  | some t =>
   cases ss with
   | nil => simp [forestWork.eq_def,GenericForest.sourceWork.eq_def,overlap,Rat.natCast_add,Rat.add_zero]
   | cons s tail =>
    cases tail with
    | nil => simp [forestWork.eq_def,GenericForest.sourceWork.eq_def,Rat.natCast_add,Rat.add_zero]
    | cons u tail =>
     rw [forestWork.eq_def,GenericForest.sourceWork.eq_def,Rat.natCast_add,Rat.add_zero]
     congr 1
     split
     · cases hover:overlap t (s::u::tail) with
       | none => simp only [hover];rfl
       | some h => simpa only [hover] using child t h
     · rfl
end OrdinaryDispatch.WorkRecursion
