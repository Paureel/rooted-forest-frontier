def OrdinaryDispatch.GlobalSourceAssembly.closedLocalPolynomial : Nat → Nat → Nat :=
fun N Q =>
  OrdinaryDispatch.FullSourceWork.prefixCoefficient N + OrdinaryDispatch.GlobalSourceAssembly.commonCoefficient N +
        OrdinaryDispatch.GlobalSourceAssembly.simpleNatCoefficient N Q +
      OrdinaryDispatch.FullSourceWork.rule4SourcePolynomial N Q +
    OrdinaryDispatch.Hard2SourceWork.hard2FrontCoefficient (2 * N + 3)
def OrdinaryDispatch.GlobalSourceAssembly.closedQueryPolynomial : Nat → Nat :=
fun n => OrdinaryDispatch.GlobalSourceAssembly.closedLocalPolynomial n n * 1024 * (n + 1)
def OrdinaryDispatch.GlobalSourceAssembly.partialOptimizeCoefficient : Nat → Nat :=
fun n =>
  n * OrdinaryDispatch.GlobalSourceAssembly.closedQueryPolynomial n + 512 * (2 * n + 1) ^ 2 +
    OrdinaryDispatch.RecoveryLocalWork.recoverySourceEnvelope n
