import RMAF
import GroupRestriction
import GroupEmbedding
import GroupPartition
import GroupCost
import GroupForward
import Geometry
import Bridge
import Composition
import GroupReconstruction
import GroupState
import GroupStateFacts
import GroupControls
import GroupDefs
import GroupForwardProof
import GroupReverseProof
import GroupCostProof
import GroupPriority
import GroupConsumerControls
import FreshReadout
import FreshConsumer
import FreshFalseTwins
import FreshChild
import Lean
open Lean Elab Command
run_cmd do
  let env ← getEnv
  let wanted : Array Name := #[`RMAF, `GroupRestriction, `GroupEmbedding, `GroupPartition, `GroupCost, `GroupForward, `Geometry, `Bridge, `Composition, `GroupReconstruction, `GroupState, `GroupStateFacts, `GroupControls, `GroupDefs, `GroupForwardProof, `GroupReverseProof, `GroupCostProof, `GroupPriority, `GroupConsumerControls, `FreshReadout, `FreshConsumer, `FreshFalseTwins, `FreshChild]
  let mut seen : Array Name := #[]
  for (n, info) in (env.constants.map₁.toList ++ env.constants.map₂.toList) do
    if let some idx := env.getModuleIdxFor? n then
      let origin := env.header.moduleNames[idx.toNat]!
      if wanted.contains origin && !seen.contains n then
        seen := seen.push n
        let axioms ← collectAxioms n
        let kind := match info with
          | .axiomInfo _ => "axiom"
          | .thmInfo _ => "theorem"
          | _ => "other"
        logInfo m!"FRESH|{origin}|{n}|{kind}|{String.intercalate "," (axioms.toList.map toString)}"
  logInfo m!"FRESH_TOTAL|{seen.size}"
